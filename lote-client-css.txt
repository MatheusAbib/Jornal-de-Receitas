function obterChaveDia(data) {
  return [
    data.getFullYear(),
    String(data.getMonth() + 1).padStart(2, '0'),
    String(data.getDate()).padStart(2, '0')
  ].join('-');
}

function obterNumeroDia(chave) {
  const [ano, mes, dia] = chave.split('-').map(Number);
  return Math.floor(Date.UTC(ano, mes - 1, dia) / 86400000);
}

function selecionarEdicao(receitas, chaveDia) {
  const ordenadas = [...receitas].sort((a, b) =>
    String(a.id).localeCompare(String(b.id), 'pt-BR', { numeric: true })
  );

  if (!ordenadas.length) {
    return { principal: null, secundarias: [] };
  }

  const numeroDia = obterNumeroDia(chaveDia);
  const indice = ((numeroDia % ordenadas.length) + ordenadas.length) % ordenadas.length;
  const principal = ordenadas[indice];
  const secundarias = [];

  for (let deslocamento = 1; deslocamento < ordenadas.length; deslocamento++) {
    if (secundarias.length === 4) break;
    secundarias.push(ordenadas[(indice + deslocamento) % ordenadas.length]);
  }

  return { principal, secundarias };
}

function obterImagemUrl(imagemUrl) {
  if (!imagemUrl) return '';

  if (
    imagemUrl.startsWith('http://') ||
    imagemUrl.startsWith('https://') ||
    imagemUrl.startsWith('data:') ||
    imagemUrl.startsWith('blob:')
  ) {
    return imagemUrl;
  }

  if (imagemUrl.startsWith('/uploads/')) return imagemUrl;

  return `/uploads/${imagemUrl.replace(/^\/+/, '')}`;
}

import { buscarReceita } from "../../../services/receitaService";
import { listarFavoritos, adicionarFavorito, removerFavorito } from "../../../services/favoritoService";

import { prefetchRota } from "../../../services/prefetch";

function DetalhesReceita() {
  const { id } = useParams();
  const navigate = useNavigate();
  const { usuario } = useAuth();
  const { mostrarToast } = useToast();

  const [receita, setReceita] = useState(null);
  const [favoritos, setFavoritos] = useState([]);
  const [carregando, setCarregando] = useState(true);
  const [carregandoFav, setCarregandoFav] = useState(false);
  const [ingredientesMarcados, setIngredientesMarcados] = useState(new Set());

  useEffect(() => {
    let ativo = true;
    let requisicaoEmAndamento = false;

    setReceita(null);
    setCarregando(true);
    setIngredientesMarcados(new Set());

    async function carregar() {
      if (requisicaoEmAndamento) return;

      requisicaoEmAndamento = true;

      try {
        const data = await buscarReceita(id);

        if (!ativo) return;

        setReceita((anterior) => {
          if (JSON.stringify(anterior) === JSON.stringify(data)) {
            return anterior;
          }

          return data;
        });
      } catch (erro) {
        if (ativo) {
          console.error(erro);
        }
      } finally {
        requisicaoEmAndamento = false;

        if (ativo) {
          setCarregando(false);
        }
      }
    }

    carregar();

    const intervalo = setInterval(carregar, 10000);

    return () => {
      ativo = false;
      clearInterval(intervalo);
    };
  }, [id]);

  useEffect(() => {
    let ativo = true;

    if (!usuario) {
      setFavoritos([]);
      return;
    }

    async function carregarFavoritos() {
      try {
        const dados = await listarFavoritos();

        if (ativo) {
          setFavoritos(Array.isArray(dados) ? dados : []);
        }
      } catch (erro) {
        console.error(erro);
      }
    }

    carregarFavoritos();

    return () => {
      ativo = false;
    };
  }, [usuario]);

  function toggleIngrediente(index) {
    setIngredientesMarcados((anteriores) => {
      const atualizados = new Set(anteriores);

      if (atualizados.has(index)) {
        atualizados.delete(index);
      } else {
        atualizados.add(index);
      }

      return atualizados;
    });
  }

  async function handleToggleFavorito() {
    if (!usuario) {
      mostrarToast('Faça login para favoritar receitas!', 'info');
      return;
    }

    if (carregandoFav || !receita) return;

    const estaFavorito = favoritos.some(
      (favorito) => String(favorito) === String(receita.id)
    );

    setCarregandoFav(true);

    try {
      if (estaFavorito) {
        await removerFavorito(receita.id);

        setFavoritos((anteriores) =>
          anteriores.filter(
            (favorito) => String(favorito) !== String(receita.id)
          )
        );

        mostrarToast('Receita removida dos favoritos.', 'info');
      } else {
        await adicionarFavorito(receita.id);

        setFavoritos((anteriores) => [...anteriores, receita.id]);

        mostrarToast('Receita adicionada aos favoritos!', 'success');
      }
    } catch (erro) {
      const mensagem = extrairMensagemErro(
        erro,
        'Erro ao atualizar favorito. Tente novamente.'
      );

      mostrarToast(mensagem, 'error');
    } finally {
      setCarregandoFav(false);
    }
  }

  function imprimir() {
    window.print();
  }

  function voltarInicio() {
    if (sessionStorage.getItem('estadoReceitas')) {
      sessionStorage.setItem('restaurarReceitas', 'true');
    }

    navigate('/');
  }

  if (carregando) {
    return (
      <PageWrapper paginaAtual="detalhe">
        <main className="detalhe-container">

          <article className="detalhe-card detalhe-carregando" aria-busy="true">

                <div className="skeleton-line skeleton-badge"></div>
                <div className="skeleton-line skeleton-titulo"></div>
                <div className="skeleton-line skeleton-titulo-curto"></div>
                <div className="skeleton-line skeleton-meta"></div>

                <div className="detalhe-meta">
                  <div className="skeleton-line skeleton-meta-item"></div>
                  <div className="skeleton-line skeleton-meta-item"></div>
                  <div className="skeleton-line skeleton-meta-item"></div>
                </div>

                <div className="detalhe-info-actions">
                  <div className="skeleton-line skeleton-action"></div>
                  <div className="skeleton-line skeleton-action"></div>
                </div>
              </div>
            </div>

            <div className="detalhe-body">
              <section className="detalhe-section">
                <div className="skeleton-line skeleton-section-title"></div>

                {Array.from({ length: 5 }, (_, index) => (
                  <div
                    key={index}
                    className="skeleton-line skeleton-item"
                  ></div>
                ))}
              </section>

              <section className="detalhe-section">
                <div className="skeleton-line skeleton-section-title"></div>

                {Array.from({ length: 4 }, (_, index) => (
                  <div
                    key={index}
                    className="skeleton-line skeleton-item"
                  ></div>
                ))}
              </section>
            </div>
          </article>
        </main>
      </PageWrapper>
    );
  }

  if (!receita) {
    return (
      <PageWrapper paginaAtual="detalhe">
        <main className="detalhe-container">
          <div className="detalhe-estado-vazio">
            <span className="detalhe-estado-selo">
              Jornal de Receitas
            </span>

            <i className="pi pi-book" aria-hidden="true"></i>

            <h1>Receita não encontrada</h1>

            <p>
              Não foi possível encontrar a receita solicitada.
            </p>

            <button
              type="button"
              className="detalhe-voltar-btn"
              onClick={voltarInicio}
              onMouseEnter={() => prefetchRota('/')}
            >
              <i className="pi pi-arrow-left" aria-hidden="true"></i>
              Voltar ao início
            </button>
          </div>
        </main>
      </PageWrapper>
    );
  }

  const imagemUrl = receita.imagem
    ? receita.imagem.startsWith('http')
      ? receita.imagem
      : `/uploads/${receita.imagem}`
    : '/default-image.jpg';

  const ingredientes = receita.ingredientes
    ? receita.ingredientes.split('||').filter((item) => item.trim())
    : [];

  const passos = receita.modoPreparo
    ? receita.modoPreparo.split('||').filter((item) => item.trim())
    : [];

  const ehSalgado = receita.categoria === 'SALGADO';

  const estaFavorito = favoritos.some(
    (favorito) => String(favorito) === String(receita.id)
  );

  const totalIngredientesMarcados = ingredientes.filter(
    (_, index) => ingredientesMarcados.has(index)
  ).length;

  const quantidadePorcoes = Number(receita.porcoes);

  const textoPorcoes = receita.porcoes
    ? `${receita.porcoes} ${quantidadePorcoes === 1 ? 'porção' : 'porções'}`

                <div className="detalhe-meta-item">
                  <i className="pi pi-book" aria-hidden="true"></i>

                    <span>Categoria</span>

                    <strong

                        ehSalgado ? 'salgado' : 'doce'

                  </div>

                        />

                        <span
                          className="detalhe-ingrediente-check"
                          aria-hidden="true"
                        ></span>

                        <span className="detalhe-ingrediente-text">
                          {ingrediente}
                        </span>
                      </label>
                    </li>
                  ))}
                </ul>
              ) : (
                <p className="detalhe-sem-conteudo">
                  Nenhum ingrediente informado.
                </p>
              )}
            </section>

            <section className="detalhe-section detalhe-section-preparo">
              <div className="detalhe-section-topo">
                <span className="detalhe-section-indice">
                  Parte II
                </span>

                <span className="detalhe-section-descricao">
                  Siga o preparo passo a passo
                </span>
              </div>

              <div className="detalhe-section-header">
                <div className="detalhe-section-titulo">
                  <i className="pi pi-book" aria-hidden="true"></i>
                  <h2>Modo de preparo</h2>
                </div>

                <span className="detalhe-section-count">
                  {passos.length} {passos.length === 1 ? 'etapa' : 'etapas'}
                </span>
              </div>

              {passos.length > 0 ? (
                <ol className="detalhe-preparo">
                  {passos.map((passo, index) => (
                    <li
                      key={index}
                      className="detalhe-preparo-item"
                    >
                      <span className="detalhe-preparo-number">
                        {String(index + 1).padStart(2, '0')}
                      </span>

                      <p className="detalhe-preparo-text">
                        {passo}
                      </p>
                    </li>
                  ))}
                </ol>
              ) : (
                <p className="detalhe-sem-conteudo">
                  Nenhuma etapa de preparo informada.
                </p>
              )}
            </section>
          </div>
        </article>
      </main>
    </PageWrapper>
  );
}

    cor: 'rejeitada'
  }
};

function MinhasReceitas() {
  const navigate = useNavigate();
  const { usuario, carregando: carregandoAuth } = useAuth();
  const { mostrarToast } = useToast();

  const [aba, setAba] = useState('aprovadas');
  const [dados, setDados] = useState({
    pendentes: [],
    aprovadas: [],
    rejeitadas: []
  });
  const [carregando, setCarregando] = useState(true);
  const [excluindo, setExcluindo] = useState(null);
  const [motivoAberto, setMotivoAberto] = useState(null);
  const [excluindoAgora, setExcluindoAgora] = useState(false);
  const [verReceita, setVerReceita] = useState(null);

  async function carregar(silencioso = false) {
    if (!silencioso) setCarregando(true);

    try {
      const resultado = await minhasReceitas(silencioso);

      setDados(resultado);
      setCache(CACHE_KEY, resultado);
    } catch (e) {
      console.error(e);
    } finally {
      if (!silencioso) setCarregando(false);
    }
  }

  useEffect(() => {
    if (carregandoAuth) return;

    if (!usuario) {
      navigate('/');
      return;
    }

    const cached = getCache(CACHE_KEY);

    if (cached) {
      setDados(cached);
      setCarregando(false);
      carregar(true);
    } else {
      carregar();
    }
  }, [usuario, carregandoAuth, navigate]);

  const modalAberto = Boolean(excluindo || motivoAberto || verReceita);

  usePolling(
    () => carregar(true),
    10000,
    !carregando && !carregandoAuth && !modalAberto
  );

  async function confirmarExclusao() {
    if (!excluindo || excluindoAgora) return;

    setExcluindoAgora(true);

    try {
      await excluirReceita(excluindo.id);

      const novo = {
        pendentes: dados.pendentes.filter(r => r.id !== excluindo.id),
        aprovadas: dados.aprovadas.filter(r => r.id !== excluindo.id),
        rejeitadas: dados.rejeitadas.filter(r => r.id !== excluindo.id)
      };

      setDados(novo);
      setCache(CACHE_KEY, novo);

      mostrarToast('Receita excluída com sucesso!', 'success');
      setExcluindo(null);
    } catch (err) {
      const msg = extrairMensagemErro(
        err,
        'Erro ao excluir receita. Tente novamente.'
      );

      mostrarToast(msg, 'error');
    } finally {
      setExcluindoAgora(false);
    }
  }

  const carregandoPagina = carregandoAuth || carregando;

  const pendentes = dados.pendentes || [];
  const aprovadas = dados.aprovadas || [];
  const rejeitadas = dados.rejeitadas || [];

  const totalReceitas =
    pendentes.length + aprovadas.length + rejeitadas.length;

  const totalCurtidas = [
    ...pendentes,
    ...aprovadas,
    ...rejeitadas
  ].reduce((acc, receita) => acc + (receita.totalFavoritos || 0), 0);

  const stats = [
    {
      icone: 'pi-book',
      valor: totalReceitas,
      label: 'Total de receitas',
      cor: 'total'
    },
    {
      icone: 'pi-check-circle',
      valor: aprovadas.length,
      label: 'Publicadas',
      cor: 'aprovadas'
    },
    {
      icone: 'pi-clock',
      valor: pendentes.length,
      label: 'Em análise',
      cor: 'pendentes'
    },
    {
      icone: 'pi-times-circle',
      valor: rejeitadas.length,
      label: 'Rejeitadas',
      cor: 'rejeitadas'
    },
    {
      icone: 'pi-heart',
      valor: totalCurtidas,
      label: 'Curtidas',
      cor: 'curtidas'
    }
  ];

  const abaAtual = ABAS.find(item => item.id === aba);
  const lista = dados[aba] || [];
  const status = STATUS[aba];

  return (
    <PageWrapper paginaAtual="minhas-receitas">
      <main className="minhas-wrapper">
        <header className="minhas-page-header">

        <div className="minhas-page-header-top">
          <div className="minhas-header-identidade">
            <span className="minhas-header-marca">JR</span>
            <span>Jornal de Receitas</span>
          </div>

          <span>Área do colaborador</span>
        </div>


          <div className="minhas-page-header-content">
            <div>
              <span className="minhas-eyebrow">Seu espaço no jornal</span>

              <h1>Minhas receitas</h1>

              <p>
                Acompanhe suas publicações, consulte o andamento
                das receitas enviadas e veja os resultados de cada uma.
              </p>
            </div>

            <button
              type="button"
              className="minhas-nova-btn"
              onClick={() => navigate('/nova')}
            >
              <i className="pi pi-plus" aria-hidden="true"></i>
              Nova receita
            </button>
          </div>
        </header>

        <section className="minhas-resumo" aria-label="Resumo das receitas">
          <div className="minhas-section-heading">
            <div>
              <span className="minhas-section-index">01 / Resumo</span>
              <h2>Visão geral</h2>
            </div>
          </div>

          <div className="minhas-stats">
            {carregandoPagina
              ? Array.from({ length: 5 }).map((_, index) => (
                  <div key={index} className="stat-card" aria-hidden="true">
                    <div className="stat-icon skeleton-icon"></div>

                    <div className="stat-info">
                      <div className="skeleton-line skeleton-line-value"></div>
                      <div className="skeleton-line skeleton-line-label"></div>
                    </div>
                  </div>
                ))
              : stats.map(item => (
                  <div key={item.cor} className={`stat-card stat-${item.cor}`}>
                    <div className={`stat-icon ${item.cor}`}>
                      <i className={`pi ${item.icone}`} aria-hidden="true"></i>
                    </div>

                    <div className="stat-info">
                      <span className="stat-value">{item.valor}</span>
                      <span className="stat-label">{item.label}</span>
                    </div>
                  </div>
                ))}
          </div>
        </section>

        <section className="minhas-listagem" aria-label="Lista de receitas">
          <div className="minhas-section-heading">
            <div>
              <span className="minhas-section-index">02 / Acompanhamento</span>
              <h2>Suas publicações</h2>
            </div>
          </div>

          <div className="minhas-tabs" role="tablist" aria-label="Status das receitas">
            {ABAS.map(item => (
              <button
                key={item.id}
                id={`minhas-tab-${item.id}`}
                type="button"
                role="tab"
                aria-selected={aba === item.id}
                aria-controls="minhas-tab-panel"
                className={`minhas-tab ${aba === item.id ? 'active' : ''}`}
                onClick={() => setAba(item.id)}
                disabled={carregandoPagina}
              >
                <i className={`pi ${item.icone}`} aria-hidden="true"></i>

                <span>{item.titulo}</span>

                {!carregandoPagina && (
                  <span className="tab-count">
                    {dados[item.id]?.length || 0}
                  </span>
                )}
              </button>
            ))}
          </div>

          <div
            id="minhas-tab-panel"
            className="minhas-tab-panel"
            role="tabpanel"
            aria-labelledby={`minhas-tab-${aba}`}
          >
            <div className="minhas-list-header">
              <div>
                <h3>{abaAtual.titulo}</h3>
                <p>{abaAtual.descricao}</p>
              </div>

              {!carregandoPagina && (
                <span className="minhas-result-count">
                  {lista.length} {lista.length === 1 ? 'receita' : 'receitas'}
                </span>
              )}
            </div>

            {carregandoPagina ? (
              <div className="minhas-grid" aria-label="Carregando receitas">
                {Array.from({ length: 6 }).map((_, index) => (
                  <SkeletonCard key={index} />
                ))}
              </div>
            ) : lista.length > 0 ? (
              <div className="minhas-grid">
                {lista.map(receita => {
                  const imagemUrl = receita.imagem
                    ? receita.imagem.startsWith('http')
                      ? receita.imagem
                      : `/uploads/${receita.imagem}`
                    : '/default-image.jpg';

                  return (
                    <article key={receita.id} className="minha-card">
                      <div className="minha-imagem">
                        <ImagemLoader
                          src={imagemUrl}
                          alt={receita.titulo}
                        />

                        <span className={`status-badge ${status.cor}`}>
                          <i
                            className={`pi ${status.icone}`}
                            aria-hidden="true"
                          ></i>
                          {status.texto}
                        </span>
                      </div>

                      <div className="minha-info">
                        <div className="minha-info-top">
                          <span className="minha-card-label">
                            <i
                              className="pi pi-book"
                              aria-hidden="true"
                            ></i>
                            Receita enviada
                          </span>

                          <h3>{receita.titulo}</h3>
                        </div>

                        <div className="minha-meta">
                          <span title="Tempo de preparo">
                            <i className="pi pi-clock" aria-hidden="true"></i>
                            {receita.tempoPreparo}
                          </span>

                          <span title="Porções">
                            <i className="pi pi-users" aria-hidden="true"></i>
                            {receita.porcoes} porções
                          </span>

                          <span title="Curtidas">
                            <i className="pi pi-heart" aria-hidden="true"></i>
                            {receita.totalFavoritos || 0} curtidas
                          </span>
                        </div>

                        <div className="minha-actions">
                          <button
                            type="button"
                            className="btn-ver"
                            onClick={() => setVerReceita(receita)}
                          >
                            <i className="pi pi-eye" aria-hidden="true"></i>
                            Ver receita
                          </button>

                          {aba === 'rejeitadas' && (
                            <button
                              type="button"
                              className="btn-motivo"
                              onClick={() => setMotivoAberto(receita)}
                            >
                              <i
                                className="pi pi-info-circle"
                                aria-hidden="true"
                              ></i>
                              Ver motivo
                            </button>
                          )}

                          {(aba === 'pendentes' || aba === 'rejeitadas') && (
                            <button
                              type="button"
                              className="btn-excluir"
                              onClick={() => setExcluindo(receita)}
                            >
                              <i className="pi pi-trash" aria-hidden="true"></i>
                              Excluir
                            </button>
                          )}
                        </div>
                      </div>
                    </article>
                  );
                })}
              </div>
            ) : (
              <div className="minhas-empty">
                <div className="minhas-empty-icon">
                  <i className="pi pi-inbox" aria-hidden="true"></i>
                </div>

                <h3>{abaAtual.vazioTitulo}</h3>
                <p>{abaAtual.vazioDescricao}</p>

                {aba !== 'rejeitadas' && (
                  <button
                    type="button"
                    className="minhas-empty-action"
                    onClick={() => navigate('/nova-receita')}
                  >
                    <i className="pi pi-plus" aria-hidden="true"></i>
                    Enviar uma receita
                  </button>
                )}
              </div>
            )}
          </div>
        </section>
      </main>

      <Modal
        aberto={!!motivoAberto}
        onFechar={() => setMotivoAberto(null)}
        maxWidth="480px"
      >
        <Modal.Header
          titulo="Motivo da rejeição"
          icone="pi pi-info-circle"
          onFechar={() => setMotivoAberto(null)}
        />

        <Modal.Body>
          <p className="motivo-texto">
            {motivoAberto?.motivoRejeicao || 'Sem motivo informado.'}
          </p>

          <div className="modal-actions">
            <button
              type="button"
              className="modal-btn modal-btn-cancel"
              onClick={() => setMotivoAberto(null)}
            >
              <i className="pi pi-times" aria-hidden="true"></i>
              Fechar
            </button>
          </div>
        </Modal.Body>
      </Modal>

      <Modal
        aberto={!!excluindo}
        onFechar={() => {
          if (!excluindoAgora) setExcluindo(null);
        }}
        maxWidth="480px"
      >
        <Modal.Header
          titulo="Excluir receita"
          icone="pi pi-exclamation-triangle"
          onFechar={() => {
            if (!excluindoAgora) setExcluindo(null);
          }}
        />

        <Modal.Body>
          <p className="admin-modal-delete-text">
            Tem certeza que deseja excluir{' '}
            <strong>{excluindo?.titulo}</strong>?
          </p>

          <p className="admin-modal-delete-warning">
            <i
              className="pi pi-exclamation-circle"
              aria-hidden="true"
            ></i>
            Esta ação não pode ser desfeita.
          </p>

          <div className="modal-actions">
            <button
              type="button"
              className="modal-btn modal-btn-cancel"
              onClick={() => setExcluindo(null)}
              disabled={excluindoAgora}
            >
              <i className="pi pi-times" aria-hidden="true"></i>
              Cancelar
            </button>

            <button
              type="button"
              className="modal-btn modal-btn-danger"
              onClick={confirmarExclusao}
              disabled={excluindoAgora}
            >
              {excluindoAgora ? (
                <>
                  <i
                    className="pi pi-spin pi-spinner"
                    aria-hidden="true"
                  ></i>
                  Excluindo...
                </>
              ) : (
                <>
                  <i className="pi pi-trash" aria-hidden="true"></i>
                  Excluir
                </>
              )}
            </button>
          </div>
        </Modal.Body>
      </Modal>

      <ModalVerReceita
        aberto={!!verReceita}
        onFechar={() => setVerReceita(null)}
        receita={verReceita}
      />
    </PageWrapper>
  );
}

import { useState, useEffect, useRef } from 'react';
import { useNavigate } from 'react-router-dom';
import PageWrapper from '../../../components/Global/PageWrapper/PageWrapper';
import { useAuth } from '../../../context/AuthContext';
import { useToast } from '../../../context/ToastContext';
import { criarReceita } from '../../../services/receitaService';
import { extrairMensagemErro } from '../../../services/api';
import { prefetchRota } from '../../../services/prefetch';
import './NovaReceita.css';

const UNIDADES = [
  { value: 'g', label: 'gramas (g)' },
  { value: 'kg', label: 'quilogramas (kg)' },
  { value: 'ml', label: 'mililitros (ml)' },
  { value: 'L', label: 'litros (L)' },
  { value: 'colher (chá)', label: 'colher (chá)' },
  { value: 'colher (sopa)', label: 'colher (sopa)' },
  { value: 'xicara', label: 'xícara (chá)' },
  { value: 'copo', label: 'copo (americano)' },
  { value: 'unidade', label: 'unidade(s)' },
  { value: 'pitada', label: 'pitada' },
  { value: 'fio', label: 'fio' }
];

function limparSufixo(valor) {
  return String(valor || '').replace(/\D/g, '');
}

function NovaReceita() {
  const navigate = useNavigate();
  const { usuario, carregando: carregandoAuth } = useAuth();
  const { mostrarToast } = useToast();

  const inputImagemRef = useRef(null);

  const [form, setForm] = useState({
    titulo: '',
    chefe: '',
    porcoes: '',
    categoria: ''
  });

  const [modoTempo, setModoTempo] = useState('minutos');
  const [tempoMinutos, setTempoMinutos] = useState('');
  const [tempoHoras, setTempoHoras] = useState('');
  const [tempoMinutosHoras, setTempoMinutosHoras] = useState('');
  const [ingredientes, setIngredientes] = useState([
    { quantidade: '1', unidade: 'g', nome: '' }
  ]);
  const [modoPreparo, setModoPreparo] = useState(['']);
  const [imagem, setImagem] = useState(null);
  const [preview, setPreview] = useState('');
  const [enviando, setEnviando] = useState(false);
  const [erro, setErro] = useState('');

  useEffect(() => {
    if (carregandoAuth) return;

    if (!usuario) {
      navigate('/');
      return;
    }

    setForm(prev => ({
      ...prev,
      chefe: prev.chefe || usuario.nome || ''
    }));
  }, [usuario, carregandoAuth, navigate]);

  useEffect(() => {
    if (!imagem) {
      setPreview('');
      return;
    }

    const url = URL.createObjectURL(imagem);
    setPreview(url);

    return () => URL.revokeObjectURL(url);
  }, [imagem]);

  function handleChange(e) {
    const { name, value } = e.target;

    setForm(prev => ({
      ...prev,
      [name]: value
    }));

    if (erro) setErro('');
  }

  function getTempoFormatado() {
    if (modoTempo === 'minutos') {
      const min = parseInt(limparSufixo(tempoMinutos), 10);
      return min > 0 ? `${min}min` : '';
    }

    const h = parseInt(limparSufixo(tempoHoras), 10) || 0;
    const m = parseInt(limparSufixo(tempoMinutosHoras), 10) || 0;
    const partes = [];

    if (h > 0) partes.push(`${h}h`);
    if (m > 0) partes.push(`${m}min`);

    return partes.join(' ');
  }

  const tempoFormatado = getTempoFormatado();

  function getTempoErro() {
    if (modoTempo === 'minutos') {
      const min = parseInt(limparSufixo(tempoMinutos), 10);

      if (min >= 60) {
        return 'Para tempos de 60 minutos ou mais, utilize o modo Horas.';
      }

      return '';
    }

    const h = parseInt(limparSufixo(tempoHoras), 10) || 0;
    const m = parseInt(limparSufixo(tempoMinutosHoras), 10) || 0;

    if (m >= 60) {
      return 'Os minutos devem ser menores que 60.';
    }

    if (h === 0 && m > 0) {
      return 'Sem horas? Utilize o modo Minutos.';
    }

    return '';
  }

  const tempoErro = getTempoErro();

  function toggleModoTempo() {
    if (modoTempo === 'minutos') {
      const min = parseInt(limparSufixo(tempoMinutos), 10) || 0;

      if (min > 0) {
        if (min >= 60) {
          const h = Math.floor(min / 60);
          const m = min % 60;

          setTempoHoras(String(h));
          setTempoMinutosHoras(m > 0 ? String(m) : '');
        } else {
          setTempoHoras('');
          setTempoMinutosHoras(String(min));
        }
      } else {
        setTempoHoras('');
        setTempoMinutosHoras('');
      }

      setTempoMinutos('');
      setModoTempo('horas');
    } else {
      const h = parseInt(limparSufixo(tempoHoras), 10) || 0;
      const m = parseInt(limparSufixo(tempoMinutosHoras), 10) || 0;
      const total = h * 60 + m;

      setTempoMinutos(total > 0 ? String(total) : '');
      setTempoHoras('');
      setTempoMinutosHoras('');
      setModoTempo('minutos');
    }
  }

  function handleTempoMinutosBlur() {
    const min = parseInt(limparSufixo(tempoMinutos), 10) || 0;
    setTempoMinutos(min > 0 ? String(min) : '');
  }

  function handleTempoHorasBlur() {
    const h = parseInt(limparSufixo(tempoHoras), 10) || 0;
    setTempoHoras(h > 0 ? String(h) : '');
  }

  function handleTempoMinutosHorasBlur() {
    const m = parseInt(limparSufixo(tempoMinutosHoras), 10) || 0;
    setTempoMinutosHoras(m > 0 && m < 60 ? String(m) : '');
  }

  function addIngrediente() {
    setIngredientes(prev => [
      ...prev,
      { quantidade: '', unidade: 'g', nome: '' }
    ]);
  }

  function removeIngrediente(index) {
    setIngredientes(prev =>
      prev.length > 1
        ? prev.filter((_, i) => i !== index)
        : prev
    );
  }

  function updateIngrediente(index, campo, valor) {
    setIngredientes(prev =>
      prev.map((ingrediente, i) =>
        i === index
          ? { ...ingrediente, [campo]: valor }
          : ingrediente
      )
    );
  }

  function addPasso() {
    setModoPreparo(prev => [...prev, '']);
  }

  function removePasso(index) {
    setModoPreparo(prev =>
      prev.length > 1
        ? prev.filter((_, i) => i !== index)
        : prev
    );
  }

  function updatePasso(index, valor) {
    setModoPreparo(prev =>
      prev.map((passo, i) => i === index ? valor : passo)
    );
  }

  function handleImagem(e) {
    const file = e.target.files?.[0];

    if (!file) return;

    if (!file.type.startsWith('image/')) {
      mostrarToast('Selecione um arquivo de imagem válido.', 'error');
      e.target.value = '';
      return;
    }

    setImagem(file);
    setErro('');
  }

  function handleRemoverImagem() {
    setImagem(null);

    if (inputImagemRef.current) {
      inputImagemRef.current.value = '';
    }
  }

  const ingredientesValidos = ingredientes.filter(
    item =>
      item.nome.trim() &&
      item.quantidade !== '' &&
      Number(item.quantidade) > 0
  );

  const passosValidos = modoPreparo.filter(passo => passo.trim());

  const formValido = Boolean(
    form.titulo.trim() &&
    form.chefe.trim() &&
    tempoFormatado &&
    !tempoErro &&
    form.porcoes &&
    Number(form.porcoes) >= 1 &&
    form.categoria &&
    ingredientesValidos.length > 0 &&
    passosValidos.length > 0 &&
    imagem
  );

  async function handleSubmit(e) {
    e.preventDefault();
    setErro('');

    if (!formValido || enviando) {
      if (!enviando) {
        const mensagem = 'Preencha todos os campos obrigatórios.';
        setErro(mensagem);
        mostrarToast(mensagem, 'error');
      }

      return;
    }

    setEnviando(true);

    try {
      const formData = new FormData();

      formData.append('titulo', form.titulo.trim());
      formData.append('chefe', form.chefe.trim());
      formData.append('tempoPreparo', tempoFormatado);
      formData.append('porcoes', form.porcoes);
      formData.append('categoria', form.categoria);

      ingredientesValidos.forEach(item => {
        formData.append(
          'ingredientes',
          `${item.quantidade} ${item.unidade} de ${item.nome.trim()}`
        );
      });

      passosValidos.forEach(passo => {
        formData.append('modoPreparo', passo.trim());
      });

      formData.append('imagemFile', imagem);

      await criarReceita(formData);

      mostrarToast('Receita enviada para aprovação!', 'success');

      prefetchRota('/minhas-receitas');

      setTimeout(() => {
        navigate('/minhas-receitas');
      }, 1000);
    } catch (err) {
      const mensagem = extrairMensagemErro(
        err,
        'Erro ao enviar receita. Tente novamente.'
      );

      setErro(mensagem);
      mostrarToast(mensagem, 'error');
      setEnviando(false);
    }
  }

  if (carregandoAuth) {
    return (
      <PageWrapper paginaAtual="nova">
        <div className="nova-auth-loading" role="status">
          <i className="pi pi-spin pi-spinner" aria-hidden="true"></i>
          <span>Carregando formulário...</span>
        </div>
      </PageWrapper>
    );
  }

  if (!usuario) return null;

  return (
    <PageWrapper paginaAtual="nova">
      <main className="nova-page">
        <header className="nova-page-header">

          <div className="nova-page-header-top">
            <div className="nova-header-identidade">
              <span className="nova-header-marca">JR</span>
              <span>Jornal de Receitas</span>
            </div>

            <span>Colabore com o jornal</span>
          </div>

          <div className="nova-page-header-content">
            <span className="nova-eyebrow">Compartilhe sua receita</span>

            <h1>Enviar uma receita</h1>

            <p>
              Preencha as informações abaixo para compartilhar sua
              preparação. Sua receita será analisada antes da publicação.
            </p>

            <button
              type="button"
              className="minhas-nova-btn"
              onClick={() => navigate('/minhas-receitas')}
            >
              <i className="pi pi-eye" aria-hidden="true"></i>
              Minhas Receitas
            </button>
          </div>

          <div className="nova-page-header-bottom">
            <span>
              <i className="pi pi-list-check" aria-hidden="true"></i>
              Preencha as quatro etapas
            </span>

            <span>
              <i className="pi pi-shield" aria-hidden="true"></i>
              Publicação sujeita à aprovação
            </span>
          </div>
        </header>

        <div className="nova-main-layout">
          <div className="nova-form-container">
            <form className="nova-recipe-form" onSubmit={handleSubmit}>
